""" Meddy Sales Intake — FastAPI router. Mount in main.py: from intake_routes import router as intake_router app.include_router(intake_router) Env vars: INTAKE_SECRET # signs session tokens (already set on Railway) INTAKE_PASSWORD # optional; overrides DEFAULT_PASSWORD below Auth is a single shared password for the internal sales team. No user accounts, no password hashing, no seeding step. To change the password: set INTAKE_PASSWORD on Railway (takes effect on redeploy), or edit DEFAULT_PASSWORD below and push. Zero new dependencies. Reuses the `supabase` client from database.py. """ import os import hmac import json import time import base64 import hashlib from typing import Optional, Dict, Any from fastapi import APIRouter, HTTPException, Header from pydantic import BaseModel from database import supabase router = APIRouter(prefix="/intake", tags=["intake"]) # Change this, or override with the INTAKE_PASSWORD env var on Railway. DEFAULT_PASSWORD = "meddy-intake-2026" PASSWORD = os.environ.get("INTAKE_PASSWORD") or DEFAULT_PASSWORD SECRET = os.environ.get("INTAKE_SECRET", "") TOKEN_TTL_SECONDS = 60 * 60 * 12 # 12h — a full day of visits # --------------------------------------------------------------- # Session tokens — HMAC-signed, stateless # --------------------------------------------------------------- def _b64e(raw: bytes) -> str: return base64.urlsafe_b64encode(raw).decode().rstrip("=") def _b64d(s: str) -> bytes: return base64.urlsafe_b64decode(s + "=" * (-len(s) % 4)) def make_token() -> str: if not SECRET: raise RuntimeError("INTAKE_SECRET not set") payload = {"exp": int(time.time()) + TOKEN_TTL_SECONDS} body = _b64e(json.dumps(payload).encode()) sig = _b64e(hmac.new(SECRET.encode(), body.encode(), hashlib.sha256).digest()) return f"{body}.{sig}" def read_token(token: str) -> Optional[Dict[str, Any]]: if not SECRET or not token or "." not in token: return None body, _, sig = token.partition(".") expected = _b64e(hmac.new(SECRET.encode(), body.encode(), hashlib.sha256).digest()) if not hmac.compare_digest(sig, expected): return None try: payload = json.loads(_b64d(body)) except Exception: return None if payload.get("exp", 0) < time.time(): return None return payload def require_session(authorization: Optional[str]) -> None: if not authorization or not authorization.startswith("Bearer "): raise HTTPException(status_code=401, detail="Not signed in") if not read_token(authorization[7:]): raise HTTPException(status_code=401, detail="Session expired, sign in again") # --------------------------------------------------------------- # Models # --------------------------------------------------------------- class LoginBody(BaseModel): password: str class IntakeBody(BaseModel): clinic_name: str specialty: Optional[str] = None address: Optional[str] = None city: Optional[str] = None pin_code: Optional[str] = None phone: Optional[str] = None website: Optional[str] = None maps_url: Optional[str] = None answers: Dict[str, Any] = {} status: str = "draft" # --------------------------------------------------------------- # Routes # --------------------------------------------------------------- @router.post("/login") def login(body: LoginBody): if not hmac.compare_digest(body.password, PASSWORD): raise HTTPException(status_code=401, detail="Password is incorrect") return {"token": make_token()} @router.post("/submit") def submit(body: IntakeBody, authorization: Optional[str] = Header(default=None)): require_session(authorization) res = supabase.table("clinic_intakes").insert(body.model_dump()).execute() if not res.data: raise HTTPException(status_code=500, detail="Could not save. Try again.") return {"id": res.data[0]["id"], "status": body.status} @router.put("/submit/{intake_id}") def update(intake_id: str, body: IntakeBody, authorization: Optional[str] = Header(default=None)): require_session(authorization) res = ( supabase.table("clinic_intakes") .update(body.model_dump()) .eq("id", intake_id) .execute() ) if not res.data: raise HTTPException(status_code=404, detail="Intake not found") return {"id": intake_id, "status": body.status} @router.get("/list") def list_intakes(authorization: Optional[str] = Header(default=None)): require_session(authorization) res = ( supabase.table("clinic_intakes") .select("id, clinic_name, city, specialty, status, created_at") .order("created_at", desc=True) .limit(200) .execute() ) return {"intakes": res.data or []} @router.get("/{intake_id}") def get_intake(intake_id: str, authorization: Optional[str] = Header(default=None)): require_session(authorization) res = supabase.table("clinic_intakes").select("*").eq("id", intake_id).limit(1).execute() rows = res.data or [] if not rows: raise HTTPException(status_code=404, detail="Intake not found") return rows[0]
